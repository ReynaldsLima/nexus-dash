  tenants: TenantOption[]
  activeSlug: string

export function TenantSwitcher({ role, tenants, activeSlug }: TenantSwitcherProps) {
  const router = useRouter()
  if (role !== 'super_admin') return null
  return (
    <select value={activeSlug} onChange={(e) => {
      const slug = e.target.value
      if (slug === '__manage__') router.push('/tenants')
      else router.push(`/${slug}/dashboard`)
    }} className="...">
      {tenants.map((t) => <option key={t.id} value={t.slug} disabled={!t.active}>{t.name}</option>)}
      <option value="__manage__">Gerenciar tenants…</option>
    </select>
  )
}

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser request → `proxy.ts` (Node runtime middleware) | Decides ONLY where to redirect — never the final authorization word; actual data access is enforced by RLS (Plan 02) and the layout guard below |
| Server Component (`[tenant-slug]/layout.tsx`) → `tenants` table | The new guard query is RLS-scoped (uses `createClient()`, never `service_role`) — an agency/Cliente user querying a tenant outside their access returns zero rows, matching real data access, not a separate authorization system that could drift from RLS |
